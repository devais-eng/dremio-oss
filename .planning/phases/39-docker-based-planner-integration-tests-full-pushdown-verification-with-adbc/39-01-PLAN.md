    ```java
    package com.dremio.plugins.jdbc.integration;

    import com.dremio.testcontainers.DremioContainer;
    import org.testcontainers.containers.GenericContainer;
    import org.testcontainers.containers.wait.strategy.Wait;
    import java.time.Duration;

    /**
     * Testcontainers wrapper for the dremio-oss:jdbc-test Docker image.
     *
     * Implements DremioContainer to satisfy the DremioRestrictedTestcontainersUsage
     * error-prone check. The image is pre-built locally; if not available the test
     * class uses Assume.assumeTrue to skip rather than fail.
     */
    public final class DremioJdbcContainer
        extends GenericContainer&lt;DremioJdbcContainer&gt;
        implements DremioContainer {

      private static final String IMAGE = "dremio-oss:jdbc-test";

      public DremioJdbcContainer() {
        super(IMAGE);
        withExposedPorts(9047)
            .waitingFor(
                Wait.forHttp("/apiv2/server_status")
                    .forPort(9047)
                    .forStatusCode(200)
                    .withStartupTimeout(Duration.ofMinutes(3)));
      }

      public String getDremioUrl() {
        return "http://" + getHost() + ":" + getMappedPort(9047);
      }
    }
    ```
