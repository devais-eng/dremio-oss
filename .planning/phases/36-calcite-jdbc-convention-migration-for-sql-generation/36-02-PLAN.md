**Rewrite getPhysicalOperator(PhysicalPlanCreator creator):**

```java
// 1. Resolve dialect
StoragePlugin rawPlugin = creator.getContext().getCatalogService()
    .getSource(pluginId.getName());
JdbcStoragePlugin jdbcPlugin = (rawPlugin instanceof JdbcStoragePlugin)
    ? (JdbcStoragePlugin) rawPlugin : null;
SqlDialect dialect = jdbcPlugin != null
    ? jdbcPlugin.createDialect() : PostgresqlSqlDialect.DEFAULT;

// 2. Create JdbcConvention and trait set
JdbcConvention convention = JdbcConvention.of(dialect, null,
    "DREMIO_JDBC_" + pluginId.getName());
RelOptCluster cluster = getCluster();
RelTraitSet jdbcTraitSet = cluster.traitSet().replace(convention);

// 3. Build left and right row types from projected columns + outputRowType
// The outputRowType contains the COMBINED row type of the join.
// Left fields are the first leftColumns.size() fields.
// Right fields are the remaining rightColumns.size() fields.
int leftCount = (leftColumns != null) ? leftColumns.size() : 0;
int rightCount = (rightColumns != null) ? rightColumns.size() : 0;

// Build left table row type from left columns
RelDataTypeFactory typeFactory = cluster.getTypeFactory();
List<RelDataTypeField> allFields = outputRowType.getFieldList();

// For the leaf nodes, we need the FULL table row types (all columns).
// However, we only have the projected column names, not the full table schema.
// Build row types from the projected columns and the outputRowType field types.
// Left leaf row type: one field per leftColumn, using original column names
// (not the Calcite dedup names from outputRowType).
List<RelDataType> leftFieldTypes = new ArrayList<>();
List<String> leftFieldNames = new ArrayList<>();
for (int i = 0; i < leftCount && i < allFields.size(); i++) {
  leftFieldTypes.add(allFields.get(i).getType());
  leftFieldNames.add(leftColumns.get(i).getRootSegment().getPath());
}
RelDataType leftRowType = typeFactory.createStructType(leftFieldTypes, leftFieldNames);

List<RelDataType> rightFieldTypes = new ArrayList<>();
List<String> rightFieldNames = new ArrayList<>();
for (int i = 0; i < rightCount && (leftCount + i) < allFields.size(); i++) {
  rightFieldTypes.add(allFields.get(leftCount + i).getType());
  rightFieldNames.add(rightColumns.get(i).getRootSegment().getPath());
}
RelDataType rightRowType = typeFactory.createStructType(rightFieldTypes, rightFieldNames);

// 4. Build leaf nodes
JdbcCalciteLeaf leftLeaf = new JdbcCalciteLeaf(
    cluster, jdbcTraitSet, leftRowType, leftSchema, leftTable);
JdbcCalciteLeaf rightLeaf = new JdbcCalciteLeaf(
    cluster, jdbcTraitSet, rightRowType, rightSchema, rightTable);

// 5. Build JdbcJoin
// The conditionRex references column indices in the COMBINED row type
// (left fields 0..leftCount-1, right fields leftCount..leftCount+rightCount-1).
// This is exactly what JdbcJoin expects.
RelNode joinNode;
try {
  joinNode = new JdbcRules.JdbcJoin(
      cluster, jdbcTraitSet, leftLeaf, rightLeaf,
      conditionRex, Collections.emptySet(), joinType);
} catch (org.apache.calcite.rel.InvalidRelException e) {
  throw new IOException("Invalid join for JDBC pushdown: " + e.getMessage(), e);
}

// 6. Optionally wrap with JdbcProject for column selection/aliasing
// The JdbcJoin output row type uses original column names from both sides.
// But the Dremio outputRowType may have dedup names (e.g., "department0").
// If the names differ, add a JdbcProject that renames columns.
RelNode root = joinNode;
RelDataType joinOutputType = joinNode.getRowType();
List<RelDataTypeField> joinFields = joinOutputType.getFieldList();
List<RelDataTypeField> expectedFields = outputRowType.getFieldList();
boolean needsProject = false;
if (joinFields.size() == expectedFields.size()) {
  for (int i = 0; i < joinFields.size(); i++) {
    if (!joinFields.get(i).getName().equals(expectedFields.get(i).getName())) {
      needsProject = true;
      break;
    }
  }
}
if (needsProject) {
  RexBuilder rexBuilder = cluster.getRexBuilder();
  List<RexNode> projects = new ArrayList<>();
  List<String> names = new ArrayList<>();
  for (int i = 0; i < joinFields.size(); i++) {
    projects.add(rexBuilder.makeInputRef(joinFields.get(i).getType(), i));
    names.add(expectedFields.get(i).getName());
  }
  RelDataType projRowType = typeFactory.createStructType(
      projects.stream().map(RexNode::getType)
          .collect(java.util.stream.Collectors.toList()), names);
  root = new JdbcRules.JdbcProject(cluster, jdbcTraitSet, root, projects, projRowType);
}

// 7. Render SQL via JdbcImplementor
org.apache.calcite.adapter.java.JavaTypeFactory jtf =
    (org.apache.calcite.adapter.java.JavaTypeFactory) typeFactory;
JdbcImplementor implementor = new JdbcImplementor(dialect, jtf);
SqlImplementor.Result result = implementor.implement(root);
String sql = result.asStatement().toSqlString(dialect).getSql();
```
