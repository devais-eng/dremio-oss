- `testCreateAndGet`: create("analyst", role) → get("analyst") returns role with correct roleName
- `testGetNonExistent`: get("unknown") → returns null (not exception)
- `testDuplicateRoleThrows`: create("analyst", role) twice → second throws RbacEntityAlreadyExistsException
- `testDeleteExistingRole`: create then delete → get returns null
- `testDeleteNonExistentThrows`: delete("unknown", grantStore, membershipStore) → throws RbacEntityNotFoundException

- `testListAll`: create 3 roles → listAll() returns list of size 3
- `testInputValidation_nullRoleId`: create(null, role) → throws IllegalArgumentException

**MembershipStoreTest cases (minimum):**

import org.junit.After;
import org.junit.Before;
import org.junit.Test;

public class RoleStoreTest {

  private LocalKVStoreProvider kvStoreProvider;
  private RoleStore roleStore;
  private GrantStore grantStore;
  private MembershipStore membershipStore;

  @Before
  public void setUp() throws Exception {
    kvStoreProvider = new LocalKVStoreProvider(DremioTest.CLASSPATH_SCAN_RESULT, null, true, false);
    kvStoreProvider.start();
    roleStore = new RoleStore(() -> kvStoreProvider);
    grantStore = new GrantStore(() -> kvStoreProvider);
    membershipStore = new MembershipStore(() -> kvStoreProvider);
  }

  @After
  public void tearDown() throws Exception {
    kvStoreProvider.close();
  }

  // ... test methods here
}
